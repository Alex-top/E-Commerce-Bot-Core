E-Commerce-Bot-Core/
├── core/                     # Ядро бизнес-логики (не зависит от платформ)
│   ├── __init__.py
│   ├── shop_bot.py           # Основной класс ядра
│   ├── database.py           # Работа с БД (товары, корзины, заказы)
│   └── models.py             # Pydantic или dataclass модели
│
├── adapters/                 # Адаптеры для разных платформ
│   ├── __init__.py
│   ├── console.py            # Заглушка для тестирования в консоли
│   ├── telegram.py           # Адаптер для Telegram
│   └── vk.py                 # Адаптер для VK
│
├── storage/                  # Данные и миграции
│   ├── db.sqlite             # Файл базы данных (будет создан)
│   └── migrations/           # Скрипты для изменения структуры БД
│
├── tests/                    # Тесты
│   └── test_core.py          # Юнит-тесты для ядра
│
├── scripts/                  # Вспомогательные скрипты
│   ├── seed.py               # Заполнение БД тестовыми товарами
│   └── console_runner.py     # Скрипт для запуска бота в консоли
│
├── .env                      # Переменные окружения (токены)
├── requirements.txt          # Зависимости
├── README.md                 # Описание проекта
└── bot.py                    # Точка входа для запуска (выбор адаптера)
